<p><a href="https://apps.apple.com/us/app/konjugieren/id6758258747">Konjugieren</a> is an iOS app for learning German verb conjugation. It shipped in March 2026 and received a UI overhaul in May 2026, which involved rebuilding verb, quiz, settings, and onboarding screens as elevated cards with semantic depth, in keeping with iOS&nbsp;26's design language. <script>
(function () {
    var STORAGE_KEY = "konjugieren-gallery-lang";

    var TITLE = { en: "Konjugieren: Before & After", de: "Konjugieren: Vorher & Nachher" };
    var BEFORE_LABEL = { en: "Before", de: "Vorher" };
    var AFTER_LABEL = { en: "After", de: "Nachher" };

    function applyLang(lang) {
        var imgs = document.querySelectorAll(".image-pair--bilingual .image-pair__img");
        for (var i = 0; i < imgs.length; i++) {
            var src = imgs[i].getAttribute("data-src-" + lang);
            var alt = imgs[i].getAttribute("data-alt-" + lang);
            if (src) { imgs[i].setAttribute("src", src); }
            if (alt) {
                imgs[i].setAttribute("alt", alt);
                imgs[i].setAttribute("title", alt);
            }
        }
        var caps = document.querySelectorAll(".image-pair--bilingual .image-pair__caption");
        for (var j = 0; j < caps.length; j++) {
            var text = caps[j].getAttribute("data-caption-" + lang);
            if (text) { caps[j].textContent = text; }
        }
        var btns = document.querySelectorAll(".lang-toggle__btn");
        for (var k = 0; k < btns.length; k++) {
            var btnLang = btns[k].getAttribute("data-lang-button");
            btns[k].setAttribute("aria-pressed", btnLang === lang ? "true" : "false");
        }
        var title = document.querySelector(".standalone__title");
        if (title && TITLE[lang]) { title.textContent = TITLE[lang]; }
        document.title = TITLE[lang] + " · Race Condition";
        var labels = document.querySelectorAll(".image-pair--bilingual .image-pair__label");
        for (var m = 0; m < labels.length; m++) {
            var which = labels[m].getAttribute("data-which");
            if (which === "before") { labels[m].textContent = BEFORE_LABEL[lang]; }
            if (which === "after") { labels[m].textContent = AFTER_LABEL[lang]; }
        }
        document.documentElement.setAttribute("lang", lang);
    }

    function initialLang() {
        var qs = new URLSearchParams(window.location.search).get("lang");
        if (qs === "en" || qs === "de") { return qs; }
        try {
            var saved = window.localStorage.getItem(STORAGE_KEY);
            if (saved === "en" || saved === "de") { return saved; }
        } catch (e) { /* localStorage may be blocked */ }
        var nav = (navigator.language || "").toLowerCase();
        if (nav.indexOf("de") === 0) { return "de"; }
        return "en";
    }

    document.addEventListener("DOMContentLoaded", function () {
        var lang = initialLang();
        applyLang(lang);

        var btns = document.querySelectorAll(".lang-toggle__btn");
        for (var i = 0; i < btns.length; i++) {
            btns[i].addEventListener("click", function (e) {
                var chosen = e.currentTarget.getAttribute("data-lang-button");
                applyLang(chosen);
                try { window.localStorage.setItem(STORAGE_KEY, chosen); } catch (err) { /* ignore */ }
            });
        }
    });
})();
</script>
